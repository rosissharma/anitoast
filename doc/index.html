Usage

// Import library
import { toast } from 'anitoast';

// Initialize once
toast.init({
  position: 'bottom-right',
  theme: 'dark',
  richColors: false,
  closeButton: false,
  duration: 4000,
});

// Basic toasts
toast('Hello world');
toast.success('Saved!');
toast.error('Failed');
toast.warning('Watch out');
toast.info('FYI');
toast.loading('Processing...');

// With options
toast.success('File uploaded', {
  description: 'report.pdf was uploaded successfully',
  duration: 8000,
  action: { label: 'View', onClick: () => {} },
});

// Promise toast
toast.promise(fetchData(), {
  loading: { title: 'Fetching...' },
  success: (data) => ({ title: 'Done', description: `Got ${data.length} items` }),
  error:   (err)  => ({ title: 'Failed', description: err.message }),
});

// Dismiss
const id = toast('Persistent', { duration: Infinity });
toast.dismiss(id);   // one
toast.dismiss();      // all
